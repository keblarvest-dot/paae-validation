        "import torch\n",
        "import torch.nn as nn\n",
        "import torch.optim as optim\n",
        "from torchvision import datasets, transforms\n",
        "from torch.utils.data import DataLoader\n",
        "import time\n",
        "import numpy as np\n",
        "import pandas as pd\n",
        "import matplotlib.pyplot as plt\n",
        "\n",
        "print(\"🔧 Configurando validación PAADE...\")\n",
        "\n",
        "# Modelo simplificado\n",
        "class PAADE_MLP(nn.Module):\n",
        "    def __init__(self, hidden_units):\n",
        "        super().__init__()\n",
        "        self.net = nn.Sequential(\n",
        "            nn.Flatten(),\n",
        "            nn.Linear(784, hidden_units),\n",
        "            nn.ReLU(),\n",
        "            nn.Linear(hidden_units, 10)\n",
        "        )\n",
        "\n",
        "    def forward(self, x):\n",
        "        return self.net(x)\n",
        "\n",
        "# Funciones\n",
        "def train_model(model, loader, epochs=1):\n",
        "    criterion = nn.CrossEntropyLoss()\n",
        "    optimizer = optim.Adam(model.parameters(), lr=0.001)\n",
        "    model.train()\n",
        "    start = time.time()\n",
        "    for epoch in range(epochs):\n",
        "        for x, y in loader:\n",
        "            optimizer.zero_grad()\n",
        "            loss = criterion(model(x), y)\n",
        "            loss.backward()\n",
        "            optimizer.step()\n",
        "    return time.time() - start\n",
        "\n",
        "def test_model(model, loader):\n",
        "    model.eval()\n",
        "    correct = 0\n",
        "    total = 0\n",
        "    with torch.no_grad():\n",
        "        for x, y in loader:\n",
        "            outputs = model(x)\n",
        "            _, predicted = torch.max(outputs.data, 1)\n",
        "            total += y.size(0)\n",
        "            correct += (predicted == y).sum().item()\n",
        "    return correct / total\n",
        "\n",
        "# Cargar datos MNIST\n",
        "print(\"📥 Descargando dataset MNIST...\")\n",
        "transform = transforms.ToTensor()\n",
        "train_dataset = datasets.MNIST(root='./data', train=True, download=True, transform=transform)\n",
        "test_dataset = datasets.MNIST(root='./data', train=False, download=True, transform=transform)\n",
        "train_loader = DataLoader(dataset=train_dataset, batch_size=128, shuffle=True)\n",
        "test_loader = DataLoader(dataset=test_dataset, batch_size=1000, shuffle=False)\n",
        "\n",
        "# Experimento PAADE\n",
        "print(\"🚀 Ejecutando experimento PAADE...\\n\")\n",
        "\n",
        "niveles = [0, 20, 40, 60, 80, 100]\n",
        "hidden_units = [512, 256, 128, 64, 32, 16]\n",
        "resultados = []\n",
        "\n",
        "for nivel, units in zip(niveles, hidden_units):\n",
        "    print(f\"Probando nivel {nivel} ({units} neuronas)...\")\n",
        "    model = PAADE_MLP(units)\n",
        "    tiempo = train_model(model, train_loader, epochs=1)\n",
        "    precision = test_model(model, test_loader)\n",
        "\n",
        "    # Calcular costo PAADE\n",
        "    overhead = 0.45 * nivel\n",
        "    penalizacion = max(0, (0.90 - precision) * 200)\n",
        "    costo_paae = tiempo + overhead + penalizacion\n",
        "\n",
        "    resultados.append({\n",
        "        'Nivel_Abstraccion': nivel,\n",
        "        'Neuronas': units,\n",
        "        'Tiempo(s)': round(tiempo, 3),\n",
        "        'Precision': round(precision, 4),\n",
        "        'Costo_PAADE': round(costo_paae, 3)\n",
        "    })\n",
        "\n",
        "# Crear DataFrame\n",
        "df = pd.DataFrame(resultados)\n",
        "print(\"\\n\" + \"=\"*60)\n",
        "print(\"📊 RESULTADOS:\")\n",
        "print(\"=\"*60)\n",
        "print(df.to_string(index=False))\n",
        "\n",
        "# Encontrar óptimo\n",
        "optimo = df.loc[df['Costo_PAADE'].idxmin()]\n",
        "print(f\"\\n✅ ÓPTIMO PAADE:\")\n",
        "print(f\"   Nivel: {optimo['Nivel_Abstraccion']}\")\n",
        "print(f\"   Costo: {optimo['Costo_PAADE']}\")\n",
        "print(f\"   Precisión: {optimo['Precision']:.2%}\")\n",
        "\n",
        "# Gráfico\n",
        "plt.figure(figsize=(10, 5))\n",
        "plt.plot(df['Nivel_Abstraccion'], df['Costo_PAADE'], 'o-', linewidth=2, markersize=8)\n",
        "plt.scatter([optimo['Nivel_Abstraccion']], [optimo['Costo_PAADE']],\n",
        "           color='gold', s=200, zorder=5, label=f'Óptimo: Nivel {int(optimo[\"Nivel_Abstraccion\"])}')\n",
        "plt.xlabel('Nivel de Abstracción (Simplificación)')\n",
        "plt.ylabel('Costo Total PAADE')\n",
        "plt.title('Validación Empírica del Principio PAADE')\n",
        "plt.legend()\n",
        "plt.grid(True, alpha=0.3)\n",
        "plt.tight_layout()\n",
        "plt.show()\n",
        "\n",
        "# Guardar CSV\n",
        "df.to_csv('paae_validacion_real.csv', index=False)\n",
        "print(\"\\n💾 Datos guardados en: paae_validacion_real.csv\")"
